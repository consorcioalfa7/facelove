   146→// Results tab content with loading state
   147→function TabContentWithLoading({
   148→  children,
   149→  isLoading,
   150→  skeletonType,
   151→  count = 6,
   152→}: {
   153→  children: React.ReactNode;
   154→  isLoading: boolean;
   155→  skeletonType: "story" | "genre" | "theme" | "author";
   156→  count?: number;
   157→}) {
   158→  if (isLoading) {
   159→    const SkeletonComponent = 
   160→      skeletonType === "story" ? StorySkeleton :
   161→      skeletonType === "genre" ? GenreSkeleton :
   162→      skeletonType === "author" ? AuthorSkeleton : ThemeBadgeSkeleton;
   163→
   164→    if (skeletonType === "theme") {
   165→      return (
   166→        <div className="flex flex-wrap gap-3 animate-in fade-in duration-300">
   167→          {[...Array(count)].map((_, i) => (
   168→            <ThemeBadgeSkeleton key={i} />
   169→          ))}
   170→        </div>
   171→      );
   172→    }
   173→
   174→    if (skeletonType === "story") {
   175→      return (
   176→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6 animate-in fade-in duration-300">
   177→          {[...Array(count)].map((_, i) => (
   178→            <StorySkeleton key={i} />
   179→          ))}
   180→        </div>
   181→      );
   182→    }
   183→
   184→    return (
   185→      <div className={cn(
   186→        "grid gap-4 animate-in fade-in duration-300",
   187→        skeletonType === "author" ? "grid-cols-1 sm:grid-cols-2 md:grid-cols-3" : "grid-cols-1 sm:grid-cols-2 md:grid-cols-3"
   188→      )}>
   189→        {[...Array(count)].map((_, i) => (
   190→          <SkeletonComponent key={i} />
   191→        ))}
   192→      </div>
   193→    );
   194→  }
   195→
   196→  return <>{children}</>;
   197→}

   220→    if (!searchQuery.trim() || searchQuery.length < 2) {
   221→      setResults(null);

   224→
   225→    setIsLoading(true);
