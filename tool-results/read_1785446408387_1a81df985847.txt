    92→// Fetch story data directly from database
    93→async function getStory(id: string): Promise<StoryResponse> {
    94→  try {
    95→    const story = await db.story.findUnique({
    96→      where: { id },
    97→      include: {
    98→        author: true,
    99→        genre: { select: { id: true, name: true, slug: true } },
   100→        themes: {
   101→          include: { theme: true },
   102→        },
   103→      },
   104→    });
   105→
   106→    if (!story) {
   107→      return { success: false, data: {} as StoryData, error: "Story not found" };
   108→    }
   109→
   110→    return {
   111→      success: true,
   112→      data: {
   113→        id: story.id,
   114→        title: story.title,
   115→        slug: story.slug,
   116→        description: story.description,
   117→        content: story.content,
   118→        rating: story.rating,
   119→        votesCount: story.votesCount,
   120→        readsCount: story.readsCount,
   121→        commentsCount: story.commentsCount,
   122→        publishedAt: story.publishedAt,
   123→        author: {
   124→          id: story.author.id,
   125→          name: story.author.name,
   126→          slug: story.author.slug,
   127→          bio: story.author.bio,
   128→          memberSince: story.author.memberSince as Date | null | undefined,
   129→        },
   130→        genre: story.genre,
   131→        themes: story.themes.map((st) => ({
   132→          id: st.theme.id,
   133→          name: st.theme.name,
   134→          slug: st.theme.slug,
   135→        })),
   136→      },
   137→    };
   138→  } catch (error) {
   139→    console.error("Error fetching story:", error);
   140→    return { success: false, data: {} as StoryData, error: String(error) };
   141→  }
   142→}

   144→// Get previous and next stories for navigation
   145→async function getAdjacentStories(
   146→  currentId: string,
   147→  genreId: string
   148→): Promise<{
   149→  previous: { id: string; title: string; slug: string; authorName: string } | null;
   150→  next: { id: string; title: string; slug: string; authorName: string } | null;
   151→}> {
   152→  try {
   153→    // Get stories in the same genre ordered by published date
   154→    const stories = await db.story.findMany({
   155→      where: {
   156→        publishedAt: { not: null },
   157→        genreId,
   158→      },
   159→      select: {
   160→        id: true,
   161→        title: true,
   162→        slug: true,
   163→        publishedAt: true,
   164→        author: { select: { name: true } },
   165→      },
   166→      orderBy: { publishedAt: "desc" },
   167→      take: 50, // Get enough to find neighbors
   168→    });
   169→
   170→    const currentIndex = stories.findIndex((s) => s.id === currentId);
   171→    
   172→    if (currentIndex === -1) {
   173→      return { previous: null, next: null };
   174→    }
   175→
   176→    return {
   177→      previous: currentIndex < stories.length - 1
   178→        ? {
   179→            id: stories[currentIndex + 1].id,
   180→            title: stories[currentIndex + 1].title,
   181→            slug: stories[currentIndex + 1].slug,
   182→            authorName: stories[currentIndex + 1].author.name,
   183→          }
   184→        : null,
   185→      next: currentIndex > 0
   186→        ? {
   187→            id: stories[currentIndex - 1].id,
   188→            title: stories[currentIndex - 1].title,
   189→            slug: stories[currentIndex - 1].slug,
   190→            authorName: stories[currentIndex - 1].author.name,
   191→          }
   192→        : null,
   193→    };
   194→  } catch (error) {
   195→    console.error("Error fetching adjacent stories:", error);
   196→    return { previous: null, next: null };
   197→  }
   198→}

   201→export async function generateMetadata({
   202→  params,
   203→}: StoryPageParams): Promise<Metadata> {
   204→  const { id } = await params;
   205→  const response = await getStory(id);
   206→
   207→  if (!response.success || !response.data?.title) {
   208→    return {
   209→      title: "História Não Encontrada | FaceLove",
   210→      description: "A história solicitada não foi encontrada.",
   211→    };
   212→  }
   213→
   214→  const story = response.data;
   215→  return {
   216→    title: `${story.title} - ${story.genre.name} | FaceLove`,
   217→    description:
   218→      story.description ||
   219→      `Leia "${story.title}" por ${story.author.name}. Uma história ${story.genre.name.toLowerCase()} no FaceLove.`,
   220→    keywords: [
   221→      story.title,
   222→      story.author.name,
   223→      story.genre.name,
   224→      ...story.themes.map((t) => t.name),
   225→      "história",
   226→      "leitura",
   227→    ],
   228→    openGraph: {
   229→      title: `${story.title} por ${story.author.name}`,
   230→      description:
   231→        story.description || `Uma história ${story.genre.name.toLowerCase()}`,
   232→      type: "article",
   233→      publishedTime: story.publishedAt ? new Date(story.publishedAt as string).toISOString() : undefined,
   234→      authors: [story.author.name],
   235→      tags: [story.genre.name, ...story.themes.map((t) => t.name)],
   236→    },
   237→  };
   238→}

   240→// Format date helper
   241→function formatDate(date: Date | string | null): string {
   242→  if (!date) return "N/A";
   243→  return new Intl.DateTimeFormat("pt-BR", {
   244→    day: "numeric",
   245→    month: "long",
   246→    year: "numeric",
   247→  }).format(new Date(date));
   248→}

   250→// Calculate reading time with more detail
   251→function calculateReadingTime(content: string): {
   252→  minutes: number;
   253→  words: number;
   254→  displayText: string;
   255→} {
   256→  const words = content.split(/\s+/).filter(w => w.length > 0).length;
   257→  const readingSpeed = 200; // words per minute
   258→  const minutes = Math.ceil(words / readingSpeed);
   259→  
   260→  let displayText: string;
   261→  if (minutes < 1) {
   262→    displayText = "< 1 min";
   263→  } else if (minutes === 1) {
   264→    displayText = "1 min de leitura";
   265→  } else if (minutes <= 5) {
   266→    displayText = `${minutes} mins de leitura`;
   267→  } else if (minutes <= 15) {
   268→    displayText = `${minutes} mins de leitura • Curta`;
   269→  } else if (minutes <= 30) {
   270→    displayText = `${minutes} mins de leitura • Média`;
   271→  } else {
   272→    displayText = `${minutes} mins de leitura • Longa`;
   273→  }
   274→  
   275→  return { minutes, words, displayText };
   276→}
