    93→// Fetch story data directly from database
    94→async function getStory(id: string): Promise<StoryResponse> {
    95→  try {
    96→    const story = await db.story.findUnique({
    97→      where: { id },
    98→      include: {
    99→        author: true,
   100→        genre: { select: { id: true, name: true, slug: true } },
   101→        themes: {
   102→          include: { theme: true },
   103→        },
   104→      },
   105→    });
   106→
   107→    if (!story) {
   108→      return { success: false, data: {} as StoryData, error: "Story not found" };
   109→    }
   110→
   111→    return {
   112→      success: true,
   113→      data: {
   114→        id: story.id,
   115→        title: story.title,
   116→        slug: story.slug,
   117→        description: story.description,
   118→        content: story.content,
   119→        rating: story.rating,
   120→        votesCount: story.votesCount,
   121→        readsCount: story.readsCount,
   122→        commentsCount: story.commentsCount,
   123→        publishedAt: story.publishedAt,
   124→        author: {
   125→          id: story.author.id,
   126→          name: story.author.name,
   127→          slug: story.author.slug,
   128→          bio: story.author.bio,
   129→          memberSince: story.author.memberSince as Date | null | undefined,
   130→        },
   131→        genre: story.genre,
   132→        themes: story.themes.map((st) => ({
   133→          id: st.theme.id,
   134→          name: st.theme.name,
   135→          slug: st.theme.slug,
   136→        })),
   137→      },
   138→    };
   139→  } catch (error) {
   140→    console.error("Error fetching story:", error);
   141→    return { success: false, data: {} as StoryData, error: String(error) };
   142→  }
   143→}

   145→// Get previous and next stories for navigation
   146→async function getAdjacentStories(
   147→  currentId: string,
   148→  genreId: string
   149→): Promise<{
   150→  previous: { id: string; title: string; slug: string; authorName: string } | null;
   151→  next: { id: string; title: string; slug: string; authorName: string } | null;
   152→}> {
   153→  try {
   154→    // Get stories in the same genre ordered by published date
   155→    const stories = await db.story.findMany({
   156→      where: {
   157→        publishedAt: { not: null },
   158→        genreId,
   159→      },
   160→      select: {
   161→        id: true,
   162→        title: true,
   163→        slug: true,
   164→        publishedAt: true,
   165→        author: { select: { name: true } },
   166→      },
   167→      orderBy: { publishedAt: "desc" },
   168→      take: 50, // Get enough to find neighbors
   169→    });
   170→
   171→    const currentIndex = stories.findIndex((s) => s.id === currentId);
   172→    
   173→    if (currentIndex === -1) {
   174→      return { previous: null, next: null };
   175→    }
   176→
   177→    return {
   178→      previous: currentIndex < stories.length - 1
   179→        ? {
   180→            id: stories[currentIndex + 1].id,
   181→            title: stories[currentIndex + 1].title,
   182→            slug: stories[currentIndex + 1].slug,
   183→            authorName: stories[currentIndex + 1].author.name,
   184→          }
   185→        : null,
   186→      next: currentIndex > 0
   187→        ? {
   188→            id: stories[currentIndex - 1].id,
   189→            title: stories[currentIndex - 1].title,
   190→            slug: stories[currentIndex - 1].slug,
   191→            authorName: stories[currentIndex - 1].author.name,
   192→          }
   193→        : null,
   194→    };
   195→  } catch (error) {
   196→    console.error("Error fetching adjacent stories:", error);
   197→    return { previous: null, next: null };
   198→  }
   199→}

   201→// Generate metadata for SEO
   202→export async function generateMetadata({
   203→  params,
   204→}: StoryPageParams): Promise<Metadata> {
   205→  const { id } = await params;
   206→  const response = await getStory(id);
   207→
   208→  if (!response.success || !response.data?.title) {
   209→    return {
   210→      title: "História Não Encontrada | FaceLove",
   211→      description: "A história solicitada não foi encontrada.",
   212→    };
   213→  }
   214→
   215→  const story = response.data;
   216→  return {
   217→    title: `${story.title} - ${story.genre.name} | FaceLove`,
   218→    description:
   219→      story.description ||
   220→      `Leia "${story.title}" por ${story.author.name}. Uma história ${story.genre.name.toLowerCase()} no FaceLove.`,
   221→    keywords: [
   222→      story.title,
   223→      story.author.name,
   224→      story.genre.name,
   225→      ...story.themes.map((t) => t.name),
   226→      "história",
   227→      "leitura",
   228→    ],
   229→    openGraph: {
   230→      title: `${story.title} por ${story.author.name}`,
   231→      description:
   232→        story.description || `Uma história ${story.genre.name.toLowerCase()}`,
   233→      type: "article",
   234→      publishedTime: story.publishedAt ? new Date(story.publishedAt as string).toISOString() : undefined,
   235→      authors: [story.author.name],
   236→      tags: [story.genre.name, ...story.themes.map((t) => t.name)],
   237→    },
   238→  };
   239→}

   241→// Format date helper
   242→function formatDate(date: Date | string | null): string {
   243→  if (!date) return "N/A";
   244→  return new Intl.DateTimeFormat("pt-BR", {
   245→    day: "numeric",
   246→    month: "long",
   247→    year: "numeric",
   248→  }).format(new Date(date));
   249→}

   251→// Calculate reading time with more detail
   252→function calculateReadingTime(content: string): {
   253→  minutes: number;
   254→  words: number;
   255→  displayText: string;
   256→} {
   257→  const words = content.split(/\s+/).filter(w => w.length > 0).length;
   258→  const readingSpeed = 200; // words per minute
   259→  const minutes = Math.ceil(words / readingSpeed);
   260→  
   261→  let displayText: string;
   262→  if (minutes < 1) {
   263→    displayText = "< 1 min";
   264→  } else if (minutes === 1) {
   265→    displayText = "1 min de leitura";
   266→  } else if (minutes <= 5) {
   267→    displayText = `${minutes} mins de leitura`;
   268→  } else if (minutes <= 15) {
   269→    displayText = `${minutes} mins de leitura • Curta`;
   270→  } else if (minutes <= 30) {
   271→    displayText = `${minutes} mins de leitura • Média`;
   272→  } else {
   273→    displayText = `${minutes} mins de leitura • Longa`;
   274→  }
   275→  
   276→  return { minutes, words, displayText };
   277→}
