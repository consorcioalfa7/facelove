    98→// Fetch story data directly from database
    99→async function getStory(id: string): Promise<StoryResponse> {
   100→  try {
   101→    const story = await db.story.findUnique({
   102→      where: { id },
   103→      include: {
   104→        author: true,
   105→        genre: { select: { id: true, name: true, slug: true } },
   106→        themes: {
   107→          include: { theme: true },
   108→        },
   109→      },
   110→    });
   111→
   112→    if (!story) {
   113→      return { success: false, data: {} as StoryData, error: "Story not found" };
   114→    }
   115→
   116→    return {
   117→      success: true,
   118→      data: {
   119→        id: story.id,
   120→        title: story.title,
   121→        slug: story.slug,
   122→        description: story.description,
   123→        content: story.content,
   124→        rating: story.rating,
   125→        votesCount: story.votesCount,
   126→        readsCount: story.readsCount,
   127→        commentsCount: story.commentsCount,
   128→        publishedAt: story.publishedAt,
   129→        author: {
   130→          id: story.author.id,
   131→          name: story.author.name,
   132→          slug: story.author.slug,
   133→          bio: story.author.bio,
   134→          memberSince: story.author.memberSince as Date | null | undefined,
   135→        },
   136→        genre: story.genre,
   137→        themes: story.themes.map((st) => ({
   138→          id: st.theme.id,
   139→          name: st.theme.name,
   140→          slug: st.theme.slug,
   141→        })),
   142→      },
   143→    };
   144→  } catch (error) {
   145→    console.error("Error fetching story:", error);
   146→    return { success: false, data: {} as StoryData, error: String(error) };
   147→  }
   148→}

   150→// Get previous and next stories for navigation
   151→async function getAdjacentStories(
   152→  currentId: string,
   153→  genreId: string
   154→): Promise<{
   155→  previous: { id: string; title: string; slug: string; authorName: string } | null;
   156→  next: { id: string; title: string; slug: string; authorName: string } | null;
   157→}> {
   158→  try {
   159→    // Get stories in the same genre ordered by published date
   160→    const stories = await db.story.findMany({
   161→      where: {
   162→        publishedAt: { not: null },
   163→        genreId,
   164→      },
   165→      select: {
   166→        id: true,
   167→        title: true,
   168→        slug: true,
   169→        publishedAt: true,
   170→        author: { select: { name: true } },
   171→      },
   172→      orderBy: { publishedAt: "desc" },
   173→      take: 50, // Get enough to find neighbors
   174→    });
   175→
   176→    const currentIndex = stories.findIndex((s) => s.id === currentId);
   177→    
   178→    if (currentIndex === -1) {
   179→      return { previous: null, next: null };
   180→    }
   181→
   182→    return {
   183→      previous: currentIndex < stories.length - 1
   184→        ? {
   185→            id: stories[currentIndex + 1].id,
   186→            title: stories[currentIndex + 1].title,
   187→            slug: stories[currentIndex + 1].slug,
   188→            authorName: stories[currentIndex + 1].author.name,
   189→          }
   190→        : null,
   191→      next: currentIndex > 0
   192→        ? {
   193→            id: stories[currentIndex - 1].id,
   194→            title: stories[currentIndex - 1].title,
   195→            slug: stories[currentIndex - 1].slug,
   196→            authorName: stories[currentIndex - 1].author.name,
   197→          }
   198→        : null,
   199→    };
   200→  } catch (error) {
   201→    console.error("Error fetching adjacent stories:", error);
   202→    return { previous: null, next: null };
   203→  }
   204→}

   208→  params,

   210→  const { id } = await params;

   230→      ...story.themes.map((t) => t.name),

   240→      authors: [story.author.name],
   241→      tags: [story.genre.name, ...story.themes.map((t) => t.name)],
   242→    },
   243→  };

   246→// Format date helper
   247→function formatDate(date: Date | string | null): string {
   248→  if (!date) return "N/A";
   249→  return new Intl.DateTimeFormat("pt-BR", {
   250→    day: "numeric",
   251→    month: "long",
   252→    year: "numeric",
   253→  }).format(new Date(date));
   254→}

   256→// Calculate reading time with more detail
   257→function calculateReadingTime(content: string): {
   258→  minutes: number;
   259→  words: number;
   260→  displayText: string;
   261→} {
   262→  const words = content.split(/\s+/).filter(w => w.length > 0).length;
   263→  const readingSpeed = 200; // words per minute
   264→  const minutes = Math.ceil(words / readingSpeed);
   265→  
   266→  let displayText: string;
   267→  if (minutes < 1) {
   268→    displayText = "< 1 min";
   269→  } else if (minutes === 1) {
   270→    displayText = "1 min de leitura";
   271→  } else if (minutes <= 5) {
   272→    displayText = `${minutes} mins de leitura`;
   273→  } else if (minutes <= 15) {
   274→    displayText = `${minutes} mins de leitura • Curta`;
   275→  } else if (minutes <= 30) {
   276→    displayText = `${minutes} mins de leitura • Média`;
   277→  } else {
   278→    displayText = `${minutes} mins de leitura • Longa`;
   279→  }
   280→  
   281→  return { minutes, words, displayText };
   282→}
