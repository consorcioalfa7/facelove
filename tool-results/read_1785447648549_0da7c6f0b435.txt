   146→   146→   146→   146→async function getAdjacentStories(
   147→   147→   147→   147→  currentId: string,
   148→   148→   148→   148→  genreId: string
   149→   149→   149→   149→): Promise<{
   150→   150→   150→   150→  previous: { id: string; title: string; slug: string; authorName: string } | null;
   151→   151→   151→   151→  next: { id: string; title: string; slug: string; authorName: string } | null;
   152→   152→   152→   152→}> {
   153→   153→   153→   153→  try {
   154→   154→   154→   154→    // Get stories in the same genre ordered by published date
   155→   155→   155→   155→    const stories = await db.story.findMany({
   156→   156→   156→   156→      where: {
   157→   157→   157→   157→        publishedAt: { not: null },
   158→   158→   158→   158→        genreId,
   159→   159→   159→   159→      },
   160→   160→   160→   160→      select: {
   161→   161→   161→   161→        id: true,
   162→   162→   162→   162→        title: true,
   163→   163→   163→   163→        slug: true,
   164→   164→   164→   164→        publishedAt: true,
   165→   165→   165→   165→        author: { select: { name: true } },
   166→   166→   166→   166→      },
   167→   167→   167→   167→      orderBy: { publishedAt: "desc" },
   168→   168→   168→   168→      take: 50, // Get enough to find neighbors
   169→   169→   169→   169→    });
   170→   170→   170→   170→
   171→   171→   171→   171→    const currentIndex = stories.findIndex((s) => s.id === currentId);
   172→   172→   172→   172→    
   173→   173→   173→   173→    if (currentIndex === -1) {
   174→   174→   174→   174→      return { previous: null, next: null };
   175→   175→   175→   175→    }
   176→   176→   176→   176→
   177→   177→   177→   177→    return {
   178→   178→   178→   178→      previous: currentIndex < stories.length - 1
   179→   179→   179→   179→        ? {
   180→   180→   180→   180→            id: stories[currentIndex + 1].id,
   181→   181→   181→   181→            title: stories[currentIndex + 1].title,
   182→   182→   182→   182→            slug: stories[currentIndex + 1].slug,
   183→   183→   183→   183→            authorName: stories[currentIndex + 1].author.name,
   184→   184→   184→   184→          }
   185→   185→   185→   185→        : null,
   186→   186→   186→   186→      next: currentIndex > 0
   187→   187→   187→   187→        ? {
   188→   188→   188→   188→            id: stories[currentIndex - 1].id,
   189→   189→   189→   189→            title: stories[currentIndex - 1].title,
   190→   190→   190→   190→            slug: stories[currentIndex - 1].slug,
   191→   191→   191→   191→            authorName: stories[currentIndex - 1].author.name,
   192→   192→   192→   192→          }
   193→   193→   193→   193→        : null,
   194→   194→   194→   194→    };
   195→   195→   195→   195→  } catch (error) {
   196→   196→   196→   196→    console.error("Error fetching adjacent stories:", error);
   197→   197→   197→   197→    return { previous: null, next: null };
   198→   198→   198→   198→  }
   199→   199→   199→   199→}

   251→   251→   251→   251→// Calculate reading time with more detail
   252→   252→   252→   252→function calculateReadingTime(content: string): {
   253→   253→   253→   253→  minutes: number;
   254→   254→   254→   254→  words: number;
   255→   255→   255→   255→  displayText: string;
   256→   256→   256→   256→} {
   257→   257→   257→   257→  const words = content.split(/\s+/).filter(w => w.length > 0).length;
   258→   258→   258→   258→  const readingSpeed = 200; // words per minute
   259→   259→   259→   259→  const minutes = Math.ceil(words / readingSpeed);
   260→   260→   260→   260→  
   261→   261→   261→   261→  let displayText: string;
   262→   262→   262→   262→  if (minutes < 1) {
   263→   263→   263→   263→    displayText = "< 1 min";
   264→   264→   264→   264→  } else if (minutes === 1) {
   265→   265→   265→   265→    displayText = "1 min de leitura";
   266→   266→   266→   266→  } else if (minutes <= 5) {
   267→   267→   267→   267→    displayText = `${minutes} mins de leitura`;
   268→   268→   268→   268→  } else if (minutes <= 15) {
   269→   269→   269→   269→    displayText = `${minutes} mins de leitura • Curta`;
   270→   270→   270→   270→  } else if (minutes <= 30) {
   271→   271→   271→   271→    displayText = `${minutes} mins de leitura • Média`;
   272→   272→   272→   272→  } else {
   273→   273→   273→   273→    displayText = `${minutes} mins de leitura • Longa`;
   274→   274→   274→   274→  }
   275→   275→   275→   275→  
   276→   276→   276→   276→  return { minutes, words, displayText };
   277→   277→   277→   277→}
